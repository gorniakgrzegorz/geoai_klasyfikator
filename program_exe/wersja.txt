# Zasób wersji wpisywany do pliku .exe przez PyInstallera.
#
# Po co to jest: Windows pokazuje te dane we właściwościach pliku,
# a przy podpisywaniu kodu przez SignPath.io są one wymagane. Reguły
# SignPatha każą ustawić nazwę produktu i wersję produktu na stałe
# wartości i wymuszać je przy każdym podpisie, żeby podpisany plik
# nie mógł się podszyć pod inny program.
#
# Wersje liczbowe (filevers, prodvers) muszą być czwórkami liczb.
# Przy zmianie wersji programu poprawia się je razem z polami
# tekstowymi FileVersion i ProductVersion oraz z plikiem
# instalator.iss i metadata.txt wtyczki.

VSVersionInfo(
  ffi=FixedFileInfo(
    filevers=(1, 0, 0, 0),
    prodvers=(1, 0, 0, 0),
    mask=0x3f,
    flags=0x0,
    OS=0x40004,
    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo([
      StringTable(
        '041504b0',
        [StringStruct('CompanyName', 'Grzegorz Gorniak'),
         StringStruct('FileDescription', 'Obiektowa klasyfikacja pokrycia terenu'),
         StringStruct('FileVersion', '1.0.0.0'),
         StringStruct('InternalName', 'GeoAI Klasyfikator'),
         StringStruct('LegalCopyright', 'Grzegorz Gorniak, GPL-3.0-or-later'),
         StringStruct('OriginalFilename', 'GeoAI Klasyfikator.exe'),
         StringStruct('ProductName', 'GeoAI Klasyfikator'),
         StringStruct('ProductVersion', '1.0.0.0')]
      )
    ]),
    VarFileInfo([VarStruct('Translation', [0x0415, 1200])])
  ]
)
